/** * Módulo do Agente de IA: Curadoria e Remodelação de Conteúdo Público */ class NewsCuratorAgent { constructor() { this.systemPrompt = ` Você é um jornalista sênior, editor de redação e analista político do portal 'Voz Verídica'. Sua tarefa é receber matérias públicas, comunicados ou rascunhos brutos e remodelá-los: 1. Extraia apenas os fatos verificáveis, o contexto central e os impactos relevantes. 2. Reescreva em tom analítico, direto e confiável, sem sensacionalismo ou exageros. 3. Crie um título forte, um resumo objetivo e um texto com boa fluidez editorial. 4. Atribua a categoria correta (Política, Economia, Sociedade, Opinião). 5. Priorize clareza, neutralidade e contexto para leitores que buscam informação útil. `; } normalizeCategory(rawCategory) { const category = String(rawCategory || 'Política').trim(); const valid = ['Política', 'Economia', 'Sociedade', 'Opinião']; return valid.includes(category) ? category : 'Política'; } createHeadline(title, category) { const cleanedTitle = String(title || '').replace(/(urgente|bomba|confira|exclusivo|alerta)/gi, '').trim(); const finalTitle = cleanedTitle || `Análise do cenário em ${category.toLowerCase()}`; return `ANÁLISE: ${finalTitle}`; } createSummary(text, category) { const normalizedText = String(text || '').replace(/\s+/g, ' ').trim(); const snippet = normalizedText.length > 120 ? `${normalizedText.slice(0, 120)}...` : normalizedText || 'Leitura essencial sobre os desdobramentos do tema.'; return `Análise crítica sobre os impactos e os limites da pauta em ${category.toLowerCase()}: ${snippet}`; } async processAndRewrite(rawData) { console.log(`[AGENTE IA] Processando matéria da fonte: ${rawData.fonte}`); const categoria = this.normalizeCategory(rawData.categoria); const tituloBruto = String(rawData.tituloBruto || 'Tema em debate'); const textoBruto = String(rawData.textoBruto || 'Conteúdo não informado.'); const fonte = String(rawData.fonte || 'Agência Pública'); const tituloRemodelado = this.createHeadline(tituloBruto, categoria); const resumoRemodelado = this.createSummary(textoBruto, categoria); const textoRemodelado = ` O cenário atual aponta para desdobramentos relevantes no setor de ${categoria.toLowerCase()}. Conforme dados apurados na matéria original (${fonte}), os fatos centrais demonstram que a discussão exige atenção, contexto e acompanhamento contínuo por parte da sociedade e das instituições. ${textoBruto} --- Análise Voz Verídica: a questão central não é apenas o anúncio isolado, mas o impacto real sobre direitos, regras, responsabilidades e percepção pública. O acompanhamento constante é essencial para que a informação continue útil, verificável e bem contextualizada. `.trim(); return { id: Date.now(), categoria, titulo: tituloRemodelado, resumo: resumoRemodelado, conteudo: textoRemodelado, autor: 'Agente IA / Redação Voz Verídica', fonteOriginal: fonte, data: new Date().toISOString().slice(0, 10), imagem: rawData.imagem || 'https://images.unsplash.com/photo-1541872703-74c5e44368f9?auto=format&fit=crop&w=800&q=80', destaque: false }; } } module.exports = new NewsCuratorAgent();
